import asyncio
import json
import random
import time
import websockets

# SENTINEL-X ドッキング用 WebSocket サーバー
# 実際のAPIサーバーのミドルウェアやプロキシログからWebsocketで送信可能

connected_clients = set()

async def handler(websocket):
    connected_clients.add(websocket)
    print("[SENTINEL-X] ダッシュボードが接続されました")
    try:
        async for message in websocket:
            # 受信メッセージ処理 (必要に応じて)
            pass
    except websockets.exceptions.ConnectionClosed:
        pass
    finally:
        connected_clients.remove(websocket)

async def broadcast_traffic_log(ip, path, method, status_code):
    if not connected_clients:
        return
    
    payload = json.dumps({
        "timestamp": time.strftime("%H:%M:%S") + f".{int(time.time() * 1000) % 1000:03d}",
        "ip": ip,
        "path": path,
        "method": method,
        "status_code": status_code
    })
    
    await asyncio.gather(*[client.send(payload) for client in connected_clients])

# サーバー起動メインループ

        print("SENTINEL-X WebSocket Server started on ws://localhost:8765")
        await asyncio.Future()  # run forever

if __name__ == "__main__":
